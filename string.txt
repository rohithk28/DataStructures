1. def max_occurring_char(s):
    freq={}
    
    for ch in s:
        freq[ch] = freq.get(ch, 0) + 1
    
    max_freq = 0
    result = None

    for ch in s:
        if freq[ch] > max_freq:
            max_freq = freq[ch]
            result = ch
    return result

2. def first_unique_char(s):
    freq={}
    for ch in s:
        freq[ch] = freq.get(ch, 0) + 1
    for i, val in enumerate(s):
        if freq[val] == 1:
            return i
    return -1    


3. def majority_element(nums):
    freq={}
    for num in nums:
        freq[num] = freq.get(num, 0) + 1
        if freq[num] > len(nums) // 2:
            return num
