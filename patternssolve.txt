2. Anagram
s= input()
s1 = input()

freq ={}

for ch in s:
    if ch == " ":
        continue
    if ch in freq:
        freq[ch] += 1
    else:
        freq[ch] = 1
        
is_anagram = True
for ch in s1:
    if ch ==" ":
        continue
    if ch not in freq or freq[ch] == 0:
        is_anagram = False
        break
    freq[ch] -= 1
for value in freq.values():
    if value != 0:
        is_anagram = False
        break
if is_anagram:
    print("Anagram")
else:
    print("Not Anagram")

** Sorting
1. Bubble sort , selection and insertion sort
def bubble(li,n):
    for i in range(n - 1):
        is_swapped = False
        for j in range (0, n - i - 1):
            if li[j] > li[j + 1]:
                li[j] , li[j + 1] = li[j + 1],li[j]
                is_swapped = True
        if not is_swapped:
            break
def selection(li,n):
    for i in range(n-1):
        min_idx = i
        for j in range(i + 1, n ):
            if li[j] < li[min_idx]:
                min_idx = j
        if min_idx != i:
            li[i],li[min_idx]= li[min_idx], li[i]

def insertion(li,n):
    for i in range(1,n):
        key = li[i]
        j = i - 1
        
        while j >= 0 and li[j] > key:
            li[j + 1] = li[j]
            j -= 1
        li[j + 1] = key
        

n = int(input())

for num in li:

** Two pointers approach - Pair Sum
n = int(input())
li = list(map(int,input().split()))
target=int(input())

left = 0
right = n - 1
count = 0

while (left < right):
    current_sum = li[left] + li[right]
    if current_sum == target:
        count += 1
        left += 1
        right -= 1
        
        while left < right and li[left] == li[left - 1]:
            left += 1
        while left < right and li[right] == li[right + 1]:
            right -= 1
        
        
    elif current_sum < target:
        left += 1 
    else:
        right -= 1
        
print(count)
