12. To find min_cost of people to transport
import math
n = int(input())
p = int(input())
q = int(input())

fuelcost = 75
cap_bus = 80
cap_shutt = 12

min_cost =None
max_bus = (n + cap_bus - 1) // cap_bus
for bus in range(max_bus):
    people_by_bus = bus * cap_bus
    remaining = n - people_by_bus

    if remaining < 0:
        remaining = 0
    shuttles = (remaining + cap_shutt -1 ) // cap_shutt
    total_cost = (bus * p + shuttles * q)* fuelcost
    
    if min_cost is None or total_cost < min_cost:
        min_cost = total_cost
print(min_cost)
