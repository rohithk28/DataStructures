class Node:
    def __init__(self,data):
        self.data = data
        self.next = None

class SinglyLinkedList:
    def __init__(self):
        self.head = None
    
    def add(self,data):
        newNode = Node(data)
        
        if self.head is None:
            self.head = newNode
            return
        
        curr = self.head
        
        while curr.next:
            curr = curr.next
        curr.next = newNode

    def display(self):
        curr = self.head
        while curr:
            print(curr.data,end= "->")
            curr = curr.next
        print("None")

class doublyNode:
    def __init__ (self,data):
        self.data = data
        self.prev = None
        self.next = None

class DoublyLinkedList:
    def __init__(self):
        self.head = None

    def add(self,data):
        NewNode = doublyNode(data)
        
        if self.head is None:
            self.head = NewNode
            return 
        
        curr = self.head
        while curr.next:
            curr = curr.next
        curr.next = NewNode
        NewNode.prev = curr

    def display(self):
        curr = self.head

        while curr:
            print(curr.data,end ="<->")
            curr = curr.next
        print("None")

class circularLink:
    def __init__(self,data):
        self.data = data
        self.next = None
class circularSinglyListing:
    def __init__(self):
        self.head = None
    
    def add(self,data):
        NewNode = circularLink(data)
        
        if self.head is None:
            self.head = NewNode
            NewNode.next = self.head
            return 
        
        curr = self.head
        while curr.next != self.head:
            curr = curr.next
        
        curr.next = NewNode
        NewNode.next = self.head

    def display(self):
        if self.head is None:
            return
        curr = self.head
        while True:
            print(curr.data,end = "->")
            curr = curr.next

            if curr == self.head:
                break
        print("(back to head)")

class Node:
    def __init__(self,data):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None

    def add_at_end(self,data):
        newNode = Node(data)
        
        if self.head is None:
            self.head = newNode
            return
        curr = self.head
        while curr.next:
            curr = curr.next
        curr.next = newNode    
    def add_at_begining(self,data):
        newNode = Node(data)
        
        newNode.next = self.head
        self.head = newNode
    
    def add_at_pos(self,data,pos):
        if pos < 1:
            return self.head
        if pos == 1:
            newNode = Node(data)
            newNode.next = self.head
            return newNode
        curr =self.head
        for i in range(1,pos - 1):
            if curr is None:
                return self.head
            curr = curr.next
        if curr is None:
            return self.head
        
        newNode = Node(data)
        newNode.next = curr.next
        curr.next = newNode
        return self.head

    def delete_at_begining(self):
        if self.head:
            self.head =  self.head.next
    def delete_at_end(self):
        if self.head is None:
            return
        curr = self.head
        while curr.next.next:
            curr = curr.next
        curr.next=curr.next.next
    def del_at_pos(self,pos):
        temp = self.head
        if pos == 1:
            self.head = temp.next
            return self.head
        for i in range(pos):
            prev = temp
            temp = temp.next

        prev.next = temp.next
        return self.head
        
    def rev(self):
        curr = self.head
        prev = None
        
        while curr:
            nextNode = curr.next
            curr.next = prev
            
            prev = curr
            curr = nextNode
        self.head = prev 
    def display(self):
        curr = self.head
        while curr:
            print(curr.data,end="->")
            curr = curr.next
        print("None")
