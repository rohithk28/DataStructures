n = int(input())
li = list(map(int,input().split()))

di = {}

for num in li:
    if num in di:
        di[num] += 1
    else:
        di[num] = 1

min=None
result = None

for key,value in di.items():
    if min is None or value < min:
        min = value
        result = key
    elif value == min and result > key:
        result = key
    
if result is None :
    print(-1)
else:
    print(result)

def is_prime(n):
    if n < 2:
        return False
    for i in range(2,int(n ** 0.5) +1):
        if n % i == 0:
            return False
    return True

def prime_range(n,n1):
    li=[]
    for i in range(n,n1+1):
        if is_prime(i):
            li.append(i)
    return li
            

n = int(input())
n1 = int(input())
print(*prime_range(n,n1))
